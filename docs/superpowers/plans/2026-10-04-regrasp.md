> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 加一条只看关节数据的「二次抓取」规则：夹爪开合次数超过同任务那批的常见上限加余量就报；在 RoboVAD train 上定余量和读哪一路夹爪，test 只跑一次，再用它重出六份报告。

**Architecture:** `checks/signal.py` 加两个纯函数（找夹爪通道、滞回数换档）；`checks/batch.py` 加 `regrasp_scores`（考卷和规则共用）和 `regrasp`（规则），复用 `_compare` 的分组和参照逻辑；`eval/exam_b_regrasp.py` 是考卷；命令行加 `eval b-regrasp`。判定逻辑（diagnose）不动。设计见 `docs/superpowers/specs/2026-10-04-regrasp-design.md`。

**Tech Stack:** Python 3.12、uv、pytest、numpy；数据是本地的 RoboVAD（`data/robovad`）。

---

## 冻结的选择（看 test 之前）

1. 常见上限 = 参照那批开合次数的 75 分位（`method="higher"`）；两档 0.35、0.65 不调。
2. 余量在 0 到 6 里选，读主臂还是从臂在两者里选，只在 train 上选：正常示范最多报出 5% 的前提下，抓空和掉落检出最多；打平看 AUC，再打平取余量大的。
3. 考卷参照用 train 里同任务的正常示范；train 里没有的任务拿这批自己当参照，单独报。
4. test 只跑一次，结果原样记。

```python
def gripper_channels(names: list[str], gripper_index: int | None) -> list[int]:
    """夹爪是哪几个通道：名字里有 gripper 的都算（双臂两只手各一个）；名字认不出，用读入时定的 gripper_index。"""
    found = [i for i, n in enumerate(names) if "gripper" in n.lower()]
    if found:
        return found
    return [] if gripper_index is None else [gripper_index]


def level_changes(x: np.ndarray, lo: float, hi: float, band: float) -> list[int]:
    """读数按 [lo, hi] 归一化，低于 0.5 - band 算一档、高于 0.5 + band 算另一档，夹在中间的不算；
    返回每次换到另一档的那一帧的下标。读数是 NaN 的帧跳过；hi 不比 lo 大时返回空。"""
    span = hi - lo
    if not span > 1e-9:
        return []
    z = (np.asarray(x, dtype=np.float64) - lo) / span
    changes: list[int] = []
    level = 0
    for i, v in enumerate(z):
        if v <= 0.5 - band:
            now = -1
        elif v >= 0.5 + band:
            now = 1
        else:  # 夹在中间，或者是 NaN
            continue
        if level != 0 and now != level:
            changes.append(i)
        level = now
    return changes
```

```python
def grip(cycles: int, n: int = 90) -> np.ndarray:
    """夹爪从合着开始，在 n 帧里开合 cycles 回（张开、合上各换一次档，共 2×cycles 次）。"""
```

```python
@dataclass(frozen=True)
class RegraspScore:
    """一条示范和参照那批比，多开合了几次（取多出最多的那只手）。"""

    n: int  # 参照那批有几条
    excess: float  # 开合次数减常见上限
    channel: str  # 哪只手的夹爪通道
    count: int  # 开合次数（张开、合上各算一次）
    upper: float  # 常见上限：参照那批开合次数的 regrasp_quantile 分位
    times: tuple[float, ...]  # 每次换档的时刻，秒，从示范开头算


def _grip(ep: Episode, source: str) -> np.ndarray:
    """source 是 action、主臂数据在而且宽度和从臂一样，就读主臂；否则读从臂。"""
    if source == "action" and ep.action is not None and ep.action.shape == ep.state.shape:
        return ep.action
    return ep.state


def _hand(name: str) -> str:
    low = name.lower()
    return "左手夹爪" if "left" in low else ("右手夹爪" if "right" in low else "夹爪")


def regrasp_scores(
    episodes: list[Episode], cfg: CheckConfig, reference: list[Episode] | None = None
) -> tuple[dict[str, RegraspScore], Skipped]:
    """每条能比的示范多开合了几次；没法比的写原因。「二次抓取」规则和考卷都用它。"""

    def summarize(group):
        chans = gripper_channels(group[0].state_names, group[0].gripper_index)
        if not chans:
            raise NotApplicable("认不出夹爪通道")
        stats = {}
        for c in chans:
            sig = [_grip(e, cfg.regrasp_source)[:, c] for e in group]
            lo, hi = np.nanpercentile(np.concatenate(sig), [5, 95])
            counts = [len(level_changes(s, lo, hi, cfg.grip_band)) for s in sig]
            if np.median(counts) == 0:  # 这只手基本不用
                continue
            stats[c] = (float(lo), float(hi), float(np.quantile(counts, cfg.regrasp_quantile, method="higher")))
        if not stats:
            raise NotApplicable(f"同任务 {len(group)} 条示范的夹爪基本不开合，不查二次抓取")
        return stats

    def judge(ep, stats, n):
        g = _grip(ep, cfg.regrasp_source)
        best = None
        for c, (lo, hi, upper) in stats.items():
            idx = level_changes(g[:, c], lo, hi, cfg.grip_band)
            if best is None or len(idx) - upper > best.excess:
                best = RegraspScore(n, len(idx) - upper, ep.state_names[c], len(idx), upper,
                                    tuple(i / ep.fps for i in idx))
        return best

    found, skipped = _compare(episodes, cfg, reference, summarize, judge)
    return {k: v[0] for k, v in found.items()}, skipped


def regrasp(
    episodes: list[Episode], cfg: CheckConfig, reference: list[Episode] | None = None
) -> tuple[Found, Skipped]:
    """夹爪开合次数超过同任务那批的「常见上限 + regrasp_margin」：像是抓了不止一回（二次抓取）。

    次数：读数按参照那批的 5 到 95 分位归一化，两档滞回数换档（signal.level_changes）；常见上限是参照那批
    开合次数的 regrasp_quantile 分位。时间段写整条示范（和「多余动作」一样），换档时刻写进证据。
    2026-10-04 加，见 docs/superpowers/specs/2026-10-04-regrasp-design.md。
    """
    scores, skipped = regrasp_scores(episodes, cfg, reference)
    found: Found = {}
    for ep in episodes:
        s = scores.get(ep.id)
        if s is None or s.excess <= cfg.regrasp_margin:
            continue
        shown = "、".join(f"{t:.1f}" for t in s.times[:12]) + ("......" if len(s.times) > 12 else "")
        found[ep.id] = [Finding(
            check="regrasp",
            cause=Cause.REGRASP,
            t0=0.0,
            t1=ep.duration,
            score=float(s.excess),
            evidence=(f"{_hand(s.channel)}开合了 {s.count} 次（张开、合上各算一次），同任务 {s.n} 条示范常见不超过 "
                      f"{s.upper:.0f} 次，像是抓了不止一回；换档在 {shown} 秒"),
        )]
    return found, skipped
```

```python
TYPES = ("failed_grasp", "object_drop", "collision", "placement_failure", "object_not_dropped")
NAMES = {"failed_grasp": "抓空", "object_drop": "掉落", "collision": "碰撞", "placement_failure": "放置失败",
         "object_not_dropped": "该掉没掉"}
SOURCES = ("action", "state")
MARGINS = tuple(range(7))
NORMAL_FLAG_MAX = 0.05


def _types(ep: Episode) -> set[str]:
    return {a["type"] for a in ep.meta["anomalies"]}


def _rate(k: int, n: int) -> dict:
    return {"k": k, "n": n, "ci": wilson(k, n)}


def _nan_low(x: float) -> float:
    return -1.0 if x != x else x


def _summary(episodes: list[Episode], scores: dict, margin: int) -> dict:
    rows = [ep for ep in episodes if ep.id in scores]
    flagged = {ep.id for ep in rows if scores[ep.id].excess > margin}
    normal = [ep for ep in rows if not ep.meta["anomalies"]]
    target = [ep for ep in rows if _types(ep) & set(TARGET)]
    anomalous = [ep for ep in rows if ep.meta["anomalies"]]

    def hit(eps):
        return sum(ep.id in flagged for ep in eps)

    y = np.array([1] * len(target) + [0] * len(normal), dtype=np.int8)
    s = np.array([scores[ep.id].excess for ep in target + normal], dtype=np.float64)
    return {
        "n_episodes": len(rows),
        "normal_flagged": _rate(hit(normal), len(normal)),
        "target_detected": _rate(hit(target), len(target)),
        "anomalous_detected": _rate(hit(anomalous), len(anomalous)),
        "by_type": {t: _rate(hit([e for e in anomalous if t in _types(e)]), sum(t in _types(e) for e in anomalous))
                    for t in TYPES},
        "auc": roc_auc(s, y) if len(target) and len(normal) else float("nan"),
    }


def _score(episodes: list[Episode], reference: list[Episode], cfg: CheckConfig):
    """见过的任务拿 reference 比；reference 里没有的任务拿这批自己比。"""
    seen_tasks = {ep.task for ep in reference}
    seen = [ep for ep in episodes if ep.task in seen_tasks]
    unseen = [ep for ep in episodes if ep.task not in seen_tasks]
    s1, k1 = regrasp_scores(seen, cfg, reference=reference)
    s2, k2 = regrasp_scores(unseen, cfg) if unseen else ({}, {})
    return seen, unseen, s1, s2, {**k1, **k2}


def run_regrasp_b(root: str | Path, split: str, source: str | None = None, margin: int | None = None) -> dict:
    """source、margin 不给就在这一批上选（只许在 train 上这么做）。"""
    root = Path(root)
    episodes = read_robovad(root, names=split_names(root, split)).episodes
    reference = [ep for ep in read_robovad(root, names=split_names(root, "train")).episodes if not ep.meta["anomalies"]]
    chosen_here = source is None or margin is None
    candidates = []
    if chosen_here:
        for src in SOURCES:
            seen, _, s1, _, _ = _score(episodes, reference, replace(CheckConfig(), regrasp_source=src))
            for m in MARGINS:
                r = _summary(seen, s1, m)
                candidates.append({"source": src, "margin": m, "target_detected": r["target_detected"],
                                   "normal_flagged": r["normal_flagged"], "auc": r["auc"]})
        ok = [c for c in candidates if c["normal_flagged"]["k"] <= NORMAL_FLAG_MAX * c["normal_flagged"]["n"]]
        best = max(ok or candidates, key=lambda c: (c["target_detected"]["k"], _nan_low(c["auc"]), c["margin"]))
        source, margin = best["source"], best["margin"]
    cfg = replace(CheckConfig(), regrasp_source=source)
    seen, unseen, s1, s2, skipped = _score(episodes, reference, cfg)
    return {
        "exam": "B·二次抓取（RoboVAD 按示范计；规则：夹爪开合次数超过同任务常见上限加余量）",
        "split": split,
        "source": source,
        "margin": margin,
        "quantile": cfg.regrasp_quantile,
        "band": cfg.grip_band,
        "chosen_on_this_split": chosen_here,
        "n_reference": len(reference),
        "candidates": candidates,
        "seen": _summary(seen, s1, margin),
        "unseen": _summary(unseen, s2, margin) if unseen else None,
        "unseen_tasks": sorted({ep.task for ep in unseen}),
        "skipped": dict(Counter(skipped.values())),
    }


def _fmt(r: dict) -> str:
    if not r["n"]:
        return "0/0"
    lo, hi = r["ci"]
    return f"{r['k']}/{r['n']} = {r['k'] / r['n']:.0%}（{lo:.0%} 到 {hi:.0%}）"


def format_regrasp_b(out: dict) -> str:
    how = "在这一批上选的" if out["chosen_on_this_split"] else "沿用 train 选定的"
    lines = [
        out["exam"],
        f"切分：{out['split']}；夹爪读{'主臂' if out['source'] == 'action' else '从臂'}；余量 {out['margin']}（{how}）；"
        f"常见上限取参照批开合次数的 {out['quantile']:.0%} 分位",
        f"参照：train 里同任务的正常示范，{out['n_reference']} 条",
    ]
    groups = (("seen", "train 里见过的任务（拿 train 的正常示范当参照）"),
              ("unseen", f"train 里没有的任务（{'、'.join(out['unseen_tasks'])}，拿这批自己当参照）"))
    for key, title in groups:
        r = out.get(key)
        if not r:
            continue
        lines += [
            f"\n{title}：{r['n_episodes']} 条",
            f"  抓空或掉落的示范报出：{_fmt(r['target_detected'])}",
            f"  有任何异常标注的报出：{_fmt(r['anomalous_detected'])}",
            f"  正常示范被报出：{_fmt(r['normal_flagged'])}",
            "  分异常类型：" + "；".join(f"{NAMES[t]} {_fmt(v)}" for t, v in r["by_type"].items() if v["n"]),
            f"  AUC（抓空或掉落对正常）：{r['auc']:.3f}",
        ]
    if out["candidates"]:
        lines.append("\n各组合（正常示范最多报出 5% 的前提下，抓空和掉落检出最多；打平看 AUC，再打平取余量大的）：")
        for c in out["candidates"]:
            lines.append(f"  {c['source']:6s} 余量 {c['margin']}：检出 {c['target_detected']['k']}/{c['target_detected']['n']}，"
                         f"正常报出 {c['normal_flagged']['k']}/{c['normal_flagged']['n']}，AUC {c['auc']:.3f}")
    if out["skipped"]:
        lines.append("\n没法比的：" + "；".join(f"{why}（{n} 条）" for why, n in out["skipped"].items()))
    return "\n".join(lines)
```
